{{/* FAQ Page JSON-LD schema. Builds the FAQPage data structure and serialises it with jsonify, so quotes and special characters in questions/answers can no longer break the JSON (the previous hand-written JSON relied on safeJS interpolation). Only renders when faqItems with both question and answer exist. Used in: activities/single.html, inspirations/single.html Expects: .Page (current page context) */}} {{ $faqItems := .Params.faqItems }} {{ $questions := slice }} {{ range $faqItems }} {{ $question := .question }} {{ $answer := .answer }} {{ if and $question $answer }} {{ $questions = $questions | append (dict "@type" "Question" "name" ($question | markdownify | plainify | htmlUnescape | strings.TrimSpace) "acceptedAnswer" (dict "@type" "Answer" "text" ($answer | markdownify | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace) ) ) }} {{ end }} {{ end }} {{ if gt (len $questions) 0 }} {{ $schema := dict "@context" "https://schema.org" "@type" "FAQPage" "@id" (printf "%s#faq" .Permalink) "url" (printf "%s#faq" .Permalink) "inLanguage" .Lang "isPartOf" (dict "@type" "WebPage" "@id" .Permalink) "mainEntity" $questions }} {{ end }}
